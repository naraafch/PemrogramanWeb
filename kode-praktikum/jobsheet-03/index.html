Contoh Kode (Query Data Peminjaman Buku)

SELECT b.judul, b.pengarang, b.tahun_terbit, a.nama_anggota, p.tanggal_pinjam, 
                    p.tanggal_kembali FROM peminjaman p JOIN buku b ON p.id_buku = b.id JOIN anggota a 
                    ON p.id_anggota = a.id WHERE p.status = 'dipinjam' ORDER BY p.tanggal_pinjam ASC;
